<script>
    function openFlightDetails() {
        document.getElementById('modal-details').classList.add('open');
    }
    function closeFlightDetails() {
        document.getElementById('modal-details').classList.remove('open');
    }
    function startReview() {
        closeFlightDetails();
        document.getElementById('modal-review').classList.add('open');
    }
    function closeReview() {
        document.getElementById('modal-review').classList.remove('open');
    }
    function finishReview() {
        alert("✅ Voo revisado com sucesso! O status no site será atualizado automaticamente.");
        closeReview();
    }
    function selectRadio(btn) {
        const group = btn.parentElement;
        group.querySelectorAll('.radio-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
    }
</script>
